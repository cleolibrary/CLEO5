{$CLEO .s}
{$INCLUDE_ONCE ../cleo_tester.inc}

script_name '0AAA'
test("0AAA (get_script_struct_named)", tests)
terminate_this_custom_script


function tests
    it("should get script struct by name", test1)
    return
    
    function test1
        script_name '0AAA'
        
        int thisPtr = get_this_script_struct
        assert_ptr(thisPtr)
        
        int ptr
        
        ptr = -1
        ptr = get_script_struct_named {scriptName} "pAAdHK2" // random not existing
        assert_eq(ptr, 0)
        
        ptr = -1
        ptr = get_script_struct_named {scriptName} '0AAA'
        assert_ptr(ptr)
        assert_eq(ptr, thisPtr)
    
        ptr = -1
        ptr = get_script_struct_named {scriptName} "CleoTest" // test runner
        assert_ptr(ptr)
        assert_neq(ptr, thisPtr)
                
        ptr = -1
        ptr = get_script_struct_named {scriptName} "I1Ctq@akBY" // not existing long
        assert_eq(ptr, 0)
        
        // rename this script
        script_name {name} "I1Ctq@akBY" // 10 characters, mixed case
        
        ptr = -1
        ptr = get_script_struct_named {scriptName} "I1Ctq@akBY" // longer than possible - truncaded same way as during setting
        assert_eq(ptr, thisPtr)
        
        ptr = -1
        ptr = get_script_struct_named {scriptName} "I1Ctq@ak" // truncated to max supported length
        assert_eq(ptr, thisPtr)
        
        ptr = -1
        ptr = get_script_struct_named {scriptName} "i1ctQ@Ak" // different letter cases
        assert_eq(ptr, thisPtr)
        
        ptr = -1
        ptr = get_script_struct_named {scriptName} "I1Ctq@a" // truncated to 7
        assert_eq(ptr, 0)
    end
end
