{$CLEO .s}
{$INCLUDE_ONCE ../cleo_tester.inc}

script_name '0B19'
test("0B19 (bit_xor_compound)", tests)
terminate_this_custom_script


function tests
    it("should XOR the variable with the operand", test1)
    it("should clear the variable when XORed with itself", test2)
    it("should keep the variable unchanged when XORed with zero", test3)
    return

    function test1
        int a = 0x0F0F
        int b = 0x00FF

        a ^= b
        assert_eq(a, 0x0FF0)
        assert_eq(b, 0x00FF) // the operand is only read
    end

    function test2
        int b = 0x12345678

        b ^= b
        assert_eq(b, 0)
    end

    function test3
        int a = 0x12345678

        a ^= 0
        assert_eq(a, 0x12345678)
    end
end
