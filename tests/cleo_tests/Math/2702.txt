{$CLEO .s}
{$INCLUDE_ONCE ../cleo_tester.inc}

script_name '2702'
test("2702 (clear_bit)", tests)
terminate_this_custom_script

function tests
    it("should clear bits", test1)
    return
    
    function test1
        int value = 0x12345678
        clear_bit {var_number} value {bitIndex} 0 // was not set
        assert_eq(value, 0x12345678)
        
        value = 0x12345678
        clear_bit {var_number} value {bitIndex} 3
        assert_eq(value, 0x12345670)
        
        value = 0x12345678
        clear_bit {var_number} value {bitIndex} 13 // was not set
        assert_eq(value, 0x12345678)
        
        value = 0x12345678
        clear_bit {var_number} value {bitIndex} 14
        assert_eq(value, 0x12341678)
        
        value = 0x12345678
        clear_bit {var_number} value {bitIndex} 28
        assert_eq(value, 0x02345678)
        
        value = 0x12345678
        clear_bit {var_number} value {bitIndex} 31 // was not set
        assert_eq(value, 0x12345678)
    end    
end
