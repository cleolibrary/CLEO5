{$CLEO .s}
{$INCLUDE_ONCE ../cleo_tester.inc}

script_name '0B11'
test("0B11 (bit_or)", tests)
terminate_this_custom_script


function tests
    it("should OR matching bits of both operands", test1)
    it("should handle all bits set", test2)
    it("should not change the operands", test3)
    return

    function test1
        int a = 0x0F0F
        int b = 0x00FF
        int c = 0

        c = a | b
        assert_eq(c, 0x0FFF)

        a = 0x12345678
        b = 0
        c = a | b
        assert_eq(c, 0x12345678)

        a = 0x80000000
        b = 0x00000001
        c = a | b
        assert_eq(c, 0x80000001)
    end

    function test2
        int a = 0x12345678
        int b = 0xFFFFFFFF
        int c = 0

        c = a | b
        assert_eq(c, 0xFFFFFFFF)
    end

    function test3
        int a = 0x0F0F
        int b = 0x00FF

        int c = a | b
        assert_eq(a, 0x0F0F)
        assert_eq(b, 0x00FF)
        assert_eq(c, 0x0FFF)
    end
end
