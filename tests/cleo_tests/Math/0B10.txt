{$CLEO .s}
{$INCLUDE_ONCE ../cleo_tester.inc}

script_name '0B10'
test("0B10 (bit_and)", tests)
terminate_this_custom_script


function tests
    it("should AND matching bits of both operands", test1)
    it("should AND with all bits set and all bits clear", test2)
    it("should not change the operands", test3)
    return

    function test1
        int a = 0x0F0F
        int b = 0x00FF
        int c = 0

        c = a & b
        assert_eq(c, 0x000F)

        a = 0xFFFFFFFF
        b = 0x12345678
        c = a & b
        assert_eq(c, 0x12345678)

        a = 0x80000000
        b = 0x80000001
        c = a & b
        assert_eq(c, 0x80000000)
    end

    function test2
        int a = 0x12345678
        int b = 0
        int c = 0

        b = 0xFFFFFFFF
        c = a & b
        assert_eq(c, 0x12345678)

        b = 0
        c = a & b
        assert_eq(c, 0)
    end

    function test3
        int a = 0x0F0F
        int b = 0x00FF

        int c = a & b
        assert_eq(a, 0x0F0F)
        assert_eq(b, 0x00FF)
        assert_eq(c, 0x000F)
    end
end
