{$CLEO .s}
{$INCLUDE_ONCE ../cleo_tester.inc}

script_name '0B17'
test("0B17 (bit_and_compound)", tests)
terminate_this_custom_script


function tests
    it("should AND the variable with the operand", test1)
    it("should keep the variable unchanged when ANDed with all bits set", test2)
    it("should clear the variable when ANDed with zero", test3)
    return

    function test1
        int a = 0x0F0F
        int b = 0x00FF

        a &= b
        assert_eq(a, 0x000F)
        assert_eq(b, 0x00FF) // the operand is only read
    end

    function test2
        int a = 0x12345678

        a &= 0xFFFFFFFF
        assert_eq(a, 0x12345678)
    end

    function test3
        int a = 0x12345678

        a &= 0
        assert_eq(a, 0)
    end
end
