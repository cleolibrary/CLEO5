{$CLEO .s}
{$INCLUDE_ONCE ../cleo_tester.inc}

script_name '0B1D'
test("0B1D (bit_shl_compound)", tests)
terminate_this_custom_script


function tests
    it("should shift the variable left", test1)
    it("should shift by zero", test2)
    it("should drop bits shifted out of the value", test3)
    return

    function test1
        int a = 1

        a <<= 4
        assert_eq(a, 0x00000010)

        a = 0x01000000
        a <<= 1
        assert_eq(a, 0x02000000)

        a = 1
        a <<= 31
        assert_eq(a, 0x80000000)
    end

    function test2
        int a = 0x12345678
        a <<= 0
        assert_eq(a, 0x12345678)
    end

    function test3
        int a = 0x40000000
        a <<= 1
        assert_eq(a, 0x80000000)

        a <<= 1
        assert_eq(a, 0) // the sign bit is shifted out
    end
end
