{$CLEO .s}
{$INCLUDE_ONCE ../cleo_tester.inc}

script_name '0AEE'
test("0AEE (pow)", tests)
terminate_this_custom_script

function tests
    it("should power numbers", test1)
    return
    
    function test1
        float value = pow {number} 3.0 {power} 4.0
        assert_eqf(value, 81.0)
        
        value = pow {number} 16.0 {power} 0.25
        assert_eqf(value, 2.0)
        
        value = pow {number} 3.1415 {power} 0.0
        assert_eqf(value, 1.0)
        
        value = pow {number} 3.0 {power} 1.0
        assert_eqf(value, 3.0)
    end
end
