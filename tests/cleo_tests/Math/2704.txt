{$CLEO .s}
{$INCLUDE_ONCE ../cleo_tester.inc}

script_name '2704'
test("2704 (is_truthy)", tests)
terminate_this_custom_script

function tests
    it("should be false", test1)
    it("should be true", test2)
    return
    
    function test1
    // int
        is_truthy {value} 0
        assert_result_false()
        
        int iVal = 0
        is_truthy {value} iVal
        assert_result_false()
        
    // float
        is_truthy {value} 0.0
        assert_result_false()
        
        float fVal = 0.0
        is_truthy {value} fVal
        assert_result_false()
        
    // short string
        is_truthy {value} ''
        assert_result_false()
        
        shortString ssVal = ''
        is_truthy {value} ssVal
        assert_result_false()
        
    // long string
        is_truthy {value} ""
        assert_result_false()
        
        longString lsVal = ""
        is_truthy {value} lsVal
        assert_result_false()
    end
    
    function test2
    // int
        is_truthy {value} 1
        assert_result_true()
        
        int iVal = 1
        is_truthy {value} iVal
        assert_result_true()
        
        is_truthy {value} 0x00001000
        assert_result_true()
        
        is_truthy {value} -1
        assert_result_true()
                
    // float
        float fVal = 0.000001
        is_truthy {value} fVal
        assert_result_true()
        
        is_truthy {value} 1.0
        assert_result_true()
                
        fVal = -0.000001
        is_truthy {value} fVal
        assert_result_true()
        
        is_truthy {value} 0.000001
        assert_result_true()
        
        is_truthy {value} -0.0 // it is 0x80000000
        assert_result_true()
        
    // short string
        is_truthy {value} 'a'
        assert_result_true()
        
        shortString ssVal = 'a'
        is_truthy {value} ssVal
        assert_result_true()
        
        is_truthy {value} ' '
        assert_result_true()
        
        is_truthy {value} 'null'
        assert_result_true()
        
    // long string
        is_truthy {value} "a"
        assert_result_true()
        
        longString lsVal = "a"
        is_truthy {value} lsVal
        assert_result_true()
        
        is_truthy {value} " "
        assert_result_true()
        
        is_truthy {value} "null"
        assert_result_true()
        
        is_truthy {value} "some very long testing string"
        assert_result_true()
    end    
end
