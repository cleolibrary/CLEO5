{$CLEO .s}
{$INCLUDE_ONCE ../cleo_tester.inc}

script_name '2701'
test("2701 (set_bit)", tests)
terminate_this_custom_script

function tests
    it("should set bits", test1)
    return
    
    function test1
        int value = 0x12345678
        set_bit {var_number} value {bitIndex} 0
        assert_eq(value, 0x12345679)
        
        value = 0x12345678
        set_bit {var_number} value {bitIndex} 3 // already set
        assert_eq(value, 0x12345678)
        
        value = 0x12345678
        set_bit {var_number} value {bitIndex} 13
        assert_eq(value, 0x12347678)
        
        value = 0x12345678
        set_bit {var_number} value {bitIndex} 14 // already set
        assert_eq(value, 0x12345678)
        
        value = 0x12345678
        set_bit {var_number} value {bitIndex} 28 // already set
        assert_eq(value, 0x12345678)
        
        value = 0x12345678
        set_bit {var_number} value {bitIndex} 31
        assert_eq(value, 0x92345678)
    end    
end
