{$CLEO .s}
{$INCLUDE_ONCE ../cleo_tester.inc}

script_name "0AC2" // set_play_3d_audio_stream_at_coords
test("0AC2 (set_play_3d_audio_stream_at_coords)", tests)
terminate_this_custom_script

const Test_File = "cleo:\cleo_tests\Audio\Ding.mp3"


function tests
    int stream
    before_each(@setup)
    after_each(@cleanup)

    it("should play a 3d stream at the given coords", test1)
    return


    :setup
        stream = load_3d_audio_stream Test_File
        assert_result_true()
    return


    :cleanup
        remove_audio_stream stream
        stream = 0
    return


    function test1(stream: int)
        int state = 0x666
        float x, y, z

        does_char_exist $scplayer
        assert_result_true()

        // position the sound next to the player
        get_offset_from_char_in_world_coords $scplayer offset -10.0 0.0 0.0 store_to x y z

        set_play_3d_audio_stream_at_coords stream {xyz} x y z // tested opcode
        set_audio_stream_volume stream {volume} 10.0
        set_audio_stream_state stream {action} AudioStreamAction.Play

        state = get_audio_stream_state stream
        assert_eq(state, 1) // AudioStreamState.Playing
    end
end
