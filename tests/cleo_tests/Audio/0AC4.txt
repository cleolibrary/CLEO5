{$CLEO .s}
{$INCLUDE_ONCE ../cleo_tester.inc}

script_name "0AC4" // set_play_3d_audio_stream_at_char
test("0AC4 (set_play_3d_audio_stream_at_char)", tests)
terminate_this_custom_script

const Test_File = "cleo:\cleo_tests\Audio\Ding.mp3"


function tests
    int stream
    before_each(@setup)
    after_each(@cleanup)

    it("should play a 3d stream attached to a character", test1)
    return


    :setup
        stream = load_3d_audio_stream Test_File
        assert_result_true()
    return


    :cleanup
        remove_audio_stream stream
        stream = 0
    return


    function test1(stream: int)
        int state = 0x666

        does_char_exist $scplayer
        assert_result_true()

        set_play_3d_audio_stream_at_char stream {character} $scplayer // tested opcode
        set_audio_stream_state stream {action} AudioStreamAction.Play

        state = get_audio_stream_state stream
        assert_eq(state, 1) // AudioStreamState.Playing
    end
end
