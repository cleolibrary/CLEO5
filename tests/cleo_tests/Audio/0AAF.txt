{$CLEO .s}
{$INCLUDE_ONCE ../cleo_tester.inc}

script_name "0AAF" // get_audio_stream_length
test("0AAF (get_audio_stream_length)", tests)
terminate_this_custom_script

const Test_File = "cleo:\cleo_tests\Audio\Ding.mp3"


function tests
    int stream
    before_each(@setup)
    after_each(@cleanup)

    it("should return the length in seconds", test1)
    return


    :setup
        stream = load_audio_stream Test_File
        assert_result_true()
    return


    :cleanup
        remove_audio_stream stream
        stream = 0
    return


    function test1(stream: int)
        int length = get_audio_stream_length stream // tested opcode
        assert_eq(length, 0) // Ding.mp3 is about 0.25s long, opcode returns whole seconds
    end
end
