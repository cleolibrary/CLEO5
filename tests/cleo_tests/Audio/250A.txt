{$CLEO .s}
{$INCLUDE_ONCE ../cleo_tester.inc}

script_name '250A'
test("250A (set_audio_stream_type)", tests)
terminate_this_custom_script

const Test_File = ".\Speech.mp3"

function tests
    int stream
    before_each(@setup)
    after_each(@cleanup)

    it("should change the stream type", test1)
    it("should accept every type value", test2)
    return

    :setup
        stream = load_audio_stream Test_File
        assert_result_true()
    return

    :cleanup
        remove_audio_stream stream
        stream = -1
    return

    function test1(stream: int)
        int type = -1

        set_audio_stream_type stream {type} AudioStreamType.None
        type = get_audio_stream_type stream
        assert_eq(type, 0) // AudioStreamType.None

        set_audio_stream_type stream {type} AudioStreamType.Music
        type = get_audio_stream_type stream
        assert_eq(type, 2) // AudioStreamType.Music
    end

    function test2(stream: int)
        int type = -1

        set_audio_stream_type stream {type} AudioStreamType.None
        type = get_audio_stream_type stream
        assert_eq(type, 0) // AudioStreamType.None

        set_audio_stream_type stream {type} AudioStreamType.Sfx
        type = get_audio_stream_type stream
        assert_eq(type, 1) // AudioStreamType.Sfx

        set_audio_stream_type stream {type} AudioStreamType.Music
        type = get_audio_stream_type stream
        assert_eq(type, 2) // AudioStreamType.Music

        set_audio_stream_type stream {type} AudioStreamType.UserInterface
        type = get_audio_stream_type stream
        assert_eq(type, 3) // AudioStreamType.UserInterface
    end
end
