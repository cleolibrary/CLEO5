{$CLEO .s}
{$INCLUDE_ONCE ../cleo_tester.inc}

script_name "0AC1" // load_3d_audio_stream
test("0AC1 (load_3d_audio_stream)", tests)
terminate_this_custom_script

const Test_File = "cleo:\cleo_tests\Audio\Ding.mp3"


function tests
    it("should fail to load non-existing file", test1)
    it("should load existing file", test2)
    return


    function test1
        int stream = 0x666
        stream = load_3d_audio_stream ".\invalid name.mp3" // tested opcode
        assert_result_false()
        assert_eq(stream, 0) // no stream created
    end


    function test2
        int stream = load_3d_audio_stream Test_File // tested opcode
        assert_result_true()
        assert_ptr(stream)
        remove_audio_stream stream
    end
end
