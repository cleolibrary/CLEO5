{$CLEO .s}
{$INCLUDE_ONCE ../cleo_tester.inc}

script_name "0ABC" // set_audio_stream_volume
test("0ABC (set_audio_stream_volume)", tests)
terminate_this_custom_script

const Test_File = "cleo:\cleo_tests\Audio\Ding.mp3"


function tests
    int stream
    before_each(@setup)
    after_each(@cleanup)

    it("should set the stream volume", test1)
    return


    :setup
        stream = load_audio_stream Test_File
        assert_result_true()
    return


    :cleanup
        remove_audio_stream stream
        stream = 0
    return


    function test1(stream: int)
        float volume = get_audio_stream_volume stream
        assert_eqf(volume, 1.0) // default volume

        set_audio_stream_volume stream {volume} 0.25 // tested opcode

        volume = get_audio_stream_volume stream
        assert_eqf(volume, 0.25)
    end
end
