{$CLEO .s}
{$INCLUDE_ONCE ../cleo_tester.inc}

script_name "0AAE" // remove_audio_stream
test("0AAE (remove_audio_stream)", tests)
terminate_this_custom_script

const Test_File = "cleo:\cleo_tests\Audio\Ding.mp3"


function tests
    it("should remove only the specified stream", test1)
    it("should allow loading the file again", test2)
    return


    function test1
        int streamA = load_audio_stream Test_File
        assert_result_true()
        int streamB = load_audio_stream Test_File
        assert_result_true()
        assert_neq(streamA, streamB)

        remove_audio_stream streamA // tested opcode

        // the other stream is unaffected
        int state = 0x666
        state = get_audio_stream_state streamB
        assert_eq(state, 2) // AudioStreamState.Paused

        remove_audio_stream streamB
    end


    function test2
        int stream = load_audio_stream Test_File
        assert_result_true()

        remove_audio_stream stream // tested opcode

        // the sound system stays usable afterwards
        stream = load_audio_stream Test_File
        assert_result_true()
        assert_ptr(stream)
        remove_audio_stream stream
    end
end
