{$CLEO .s}
{$INCLUDE_ONCE ../cleo_tester.inc}

script_name "0AAD" // set_audio_stream_state
test("0AAD (set_audio_stream_state)", tests)
terminate_this_custom_script

const Test_File = "cleo:\cleo_tests\Audio\Ding.mp3"


function tests
    int stream
    before_each(@setup)
    after_each(@cleanup)

    it("should pause a new stream", test1)
    it("should play the stream", test2)
    it("should stop the stream", test3)
    return


    :setup
        stream = load_audio_stream Test_File
        assert_result_true()
    return


    :cleanup
        remove_audio_stream stream
        stream = 0
    return


    function test1(stream: int)
        int state = 0x666

        set_audio_stream_state stream {action} AudioStreamAction.Pause // tested opcode
        state = get_audio_stream_state stream
        assert_eq(state, 2) // AudioStreamState.Paused
    end


    function test2(stream: int)
        int state = 0x666

        set_audio_stream_state stream {action} AudioStreamAction.Play // tested opcode
        state = get_audio_stream_state stream
        assert_eq(state, 1) // AudioStreamState.Playing
    end


    function test3(stream: int)
        int state = 0x666

        set_audio_stream_state stream {action} AudioStreamAction.Stop // tested opcode
        state = get_audio_stream_state stream
        assert_eq(state, -1) // AudioStreamState.Stopped
    end
end
