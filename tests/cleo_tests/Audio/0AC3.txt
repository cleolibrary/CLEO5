{$CLEO .s}
{$INCLUDE_ONCE ../cleo_tester.inc}

script_name "0AC3"
test("0AC3 (set_play_3d_audio_stream_at_object)", tests)
terminate_this_custom_script

const Test_File = "cleo:\cleo_tests\Audio\Ding.mp3"
const MODEL_BOMB = 364


function tests
    int stream, objectHandle
    before_each(@setup)
    after_each(@cleanup)

    it("should play a 3d stream at the given object", test1)
    return


    :setup
        stream = load_3d_audio_stream Test_File
        assert_result_true()
        
        objectHandle = SpawnObject(MODEL_BOMB)
        does_object_exist {handle} objectHandle
        assert_result_true()
    return


    :cleanup
        remove_audio_stream stream
        stream = 0
        
        delete_object objectHandle
        objectHandle = -1
    return


    function test1(stream: int, objectHandle: int)        
        set_play_3d_audio_stream_at_object stream {object} objectHandle // tested opcode
        set_audio_stream_volume stream {volume} 10.0
        set_audio_stream_state stream {action} AudioStreamAction.Play

        int state = 0x666
        state = get_audio_stream_state stream
        assert_eq(state, 1) // AudioStreamState.Playing
    end
end


function SpawnObject(modelIdx: int): int
    float x, y, z
    x = generate_random_float_in_range {min} -5.0 {max} 5.0
    y = generate_random_float_in_range {min} 4.0 {max} 6.0
    z = -1.0

    x, y, z = get_offset_from_char_in_world_coords $scplayer {offset} x y z
    
    request_model modelIdx
    load_all_models_now
    int objectHandle = create_object {modelId} modelIdx {pos} x y z
    mark_model_as_no_longer_needed modelIdx
    
    return objectHandle
end
