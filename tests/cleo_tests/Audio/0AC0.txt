{$CLEO .s}
{$INCLUDE_ONCE ../cleo_tester.inc}

script_name "0AC0" // set_audio_stream_looped
test("0AC0 (set_audio_stream_looped)", tests)
terminate_this_custom_script

const Test_File = "cleo:\cleo_tests\Audio\Ding.mp3"


function tests
    int stream
    before_each(@setup)
    after_each(@cleanup)

    it("should keep the stream playing past its length", test1)
    return


    :setup
        stream = load_audio_stream Test_File
        assert_result_true()
    return


    :cleanup
        remove_audio_stream stream
        stream = 0
    return


    function test1(stream: int)
        int state = 0x666

        set_audio_stream_looped stream {flag} true // tested opcode
        set_audio_stream_state stream {action} AudioStreamAction.Play

        state = get_audio_stream_state stream
        assert_eq(state, 1) // AudioStreamState.Playing

        wait {time} 400 // Ding.mp3 is about 0.25s long

        state = get_audio_stream_state stream
        assert_eq(state, 1) // still playing thanks to looping

        set_audio_stream_state stream {action} AudioStreamAction.Stop
    end
end
