{$CLEO .s}
{$INCLUDE_ONCE ../cleo_tester.inc}

script_name "0A9B" // close_file
test("0A9B (close_file)", tests)
terminate_this_custom_script

const Test_File = "cleo\.cleo_config.ini"


function tests
    it("should close the file and release the handle", test1)
    return


    function test1
        int fh = open_file Test_File {mode} "r+"
        assert_result_true()

        close_file fh // tested opcode

        // the handle is released: the file can be opened and closed again
        fh = open_file Test_File {mode} "r+"
        assert_result_true()

        close_file fh // tested opcode

        fh = open_file Test_File {mode} "r+"
        assert_result_true()

        close_file fh
    end
end
