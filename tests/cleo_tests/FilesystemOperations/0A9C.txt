{$CLEO .s}
{$INCLUDE_ONCE ../cleo_tester.inc}

script_name "0A9C" // get_file_size
test("0A9C (get_file_size)", tests)
terminate_this_custom_script

const Test_File = "root:\cleo.asi"


function tests
    it("should return the size of an existing file", test1)
    return


    function test1
        int fh = open_file Test_File {mode} "r"
        assert_result_true()

        int size = get_file_size fh // tested opcode
        assert_range(size, 1, 0x7FFFFFFF) // cleo.asi is never empty

        close_file fh
    end
end
