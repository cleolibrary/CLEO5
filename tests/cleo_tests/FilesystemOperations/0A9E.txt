{$CLEO .s}
{$INCLUDE_ONCE ../cleo_tester.inc}

script_name "0A9E" // write_to_file
test("0A9E (write_to_file)", tests)
terminate_this_custom_script

const Test_File = "cleo:\cleo_tests\test_file.dat"


function tests
    int fh
    before_each(@setup)
    after_each(@cleanup)

    it("should write 0 bytes", test1)
    it("should write 1 byte", test2)
    it("should write 2 bytes", test3)
    it("should write 3 bytes", test4)
    it("should write 4 bytes", test5)
    it("should write 5 bytes", test6)
    it("should write 9 bytes", test7)
    it("should write the expected data", test8)
    return


    :setup
        if
            does_file_exist {path} Test_File
        then
            delete_file {path} Test_File
        end

        fh = open_file Test_File {mode} "w+"
        assert_result_true()
    return


    :cleanup
        close_file fh
        delete_file {path} Test_File
    return


    // the source value is 0x77665544, its neighbours make up the byte sequence
    // 44 55 66 77 88 99 AA BB DD CC EE FF (little endian)


    function test1(fh: int)
        int src  = 0x77665544
        int src2 = 0xBBAA9988
        int src3 = 0xFFEECCDD

        write_to_file fh {size} 0 {var_source} src // tested opcode

        int size = get_file_size fh
        assert_eq(size, 0)
    end


    function test2(fh: int)
        int src  = 0x77665544
        int src2 = 0xBBAA9988
        int src3 = 0xFFEECCDD

        write_to_file fh {size} 1 {var_source} src // tested opcode

        int size = get_file_size fh
        assert_eq(size, 1)
    end


    function test3(fh: int)
        int src  = 0x77665544
        int src2 = 0xBBAA9988
        int src3 = 0xFFEECCDD

        write_to_file fh {size} 2 {var_source} src // tested opcode

        int size = get_file_size fh
        assert_eq(size, 2)
    end


    function test4(fh: int)
        int src  = 0x77665544
        int src2 = 0xBBAA9988
        int src3 = 0xFFEECCDD

        write_to_file fh {size} 3 {var_source} src // tested opcode

        int size = get_file_size fh
        assert_eq(size, 3)
    end


    function test5(fh: int)
        int src  = 0x77665544
        int src2 = 0xBBAA9988
        int src3 = 0xFFEECCDD

        write_to_file fh {size} 4 {var_source} src // tested opcode

        int size = get_file_size fh
        assert_eq(size, 4)
    end


    function test6(fh: int)
        int src  = 0x77665544
        int src2 = 0xBBAA9988
        int src3 = 0xFFEECCDD

        write_to_file fh {size} 5 {var_source} src // tested opcode

        int size = get_file_size fh
        assert_eq(size, 5)
    end


    function test7(fh: int)
        int src  = 0x77665544
        int src2 = 0xBBAA9988
        int src3 = 0xFFEECCDD

        write_to_file fh {size} 9 {var_source} src // tested opcode

        int size = get_file_size fh
        assert_eq(size, 9)
    end


    function test8(fh: int)
        int src  = 0x77665544
        int src2 = 0xBBAA9988
        int src3 = 0xFFEECCDD

        write_to_file fh {size} 0 {var_source} src
        write_to_file fh {size} 1 {var_source} src
        write_to_file fh {size} 2 {var_source} src
        write_to_file fh {size} 3 {var_source} src
        write_to_file fh {size} 4 {var_source} src
        write_to_file fh {size} 5 {var_source} src
        write_to_file fh {size} 9 {var_source} src

        int size = get_file_size fh
        assert_eq(size, 24) // 0 + 1 + 2 + 3 + 4 + 5 + 9

        file_seek fh {offset} 0 {origin} SeekOrigin.Begin

        int chunk = 0
        chunk = read_from_file fh {size} 4
        assert_eq(chunk, 0x44554444)
        chunk = read_from_file fh {size} 4
        assert_eq(chunk, 0x55446655)
        chunk = read_from_file fh {size} 4
        assert_eq(chunk, 0x55447766)
        chunk = read_from_file fh {size} 4
        assert_eq(chunk, 0x44887766)
    end
end
