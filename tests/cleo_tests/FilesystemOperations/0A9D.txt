{$CLEO .s}
{$INCLUDE_ONCE ../cleo_tester.inc}

script_name "0A9D" // read_from_file
test("0A9D (read_from_file)", tests)
terminate_this_custom_script

const Test_File = "root:\cleo.asi"


function tests
    int fh
    before_each(@setup)
    after_each(@cleanup)

    it("should read 0 bytes", test1)
    it("should read 2 bytes", test2)
    it("should read 2 bytes into an array element", test3)
    return


    :setup
        fh = open_file Test_File {mode} "r"
        assert_result_true()
    return


    :cleanup
        close_file fh
    return


    function test1(fh: int)
        int a = 0xcccccccc
        int b = 0xdddddddd
        int c = 0xeeeeeeee

        read_from_file fh {size} 0 {destination} b // tested opcode

        assert_eq(a, 0xcccccccc) // unchanged
        assert_eq(b, 0x00000000) // destination cleared
        assert_eq(c, 0xeeeeeeee) // unchanged
    end


    function test2(fh: int)
        int a = 0xcccccccc
        int b = 0xdddddddd
        int c = 0xeeeeeeee

        read_from_file fh {size} 2 {destination} b // tested opcode

        assert_eq(a, 0xcccccccc) // unchanged
        assert_eq(b, 0x00005A4D) // DOS "MZ" header
        assert_eq(c, 0xeeeeeeee) // unchanged
    end


    function test3(fh: int)
        int buf[3]
        buf[0] = 0xcccccccc
        buf[1] = 0xdddddddd
        buf[2] = 0xeeeeeeee

        read_from_file fh {size} 2 {destination} buf[1] // tested opcode

        assert_eq(buf[0], 0xcccccccc) // unchanged
        assert_eq(buf[1], 0x00005A4D) // DOS "MZ" header
        assert_eq(buf[2], 0xeeeeeeee) // unchanged
    end
end
