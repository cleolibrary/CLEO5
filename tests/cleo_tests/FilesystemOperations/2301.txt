{$CLEO .s}
{$INCLUDE_ONCE ../cleo_tester.inc}

script_name "2301" // read_block_from_file
test("2301 (read_block_from_file)", tests)
terminate_this_custom_script

const Test_File = "cleo:\cleo_tests\test_block.dat"


function tests
    int fh
    before_each(@setup)
    after_each(@cleanup)

    it("should read 0 bytes", test1)
    it("should read 1 byte", test2)
    it("should read 2 bytes", test3)
    it("should read 3 bytes", test4)
    it("should read 4 bytes", test5)
    it("should read 6 bytes", test6)
    return


    :setup
        if
            does_file_exist {path} Test_File
        then
            delete_file {path} Test_File
        end

        // file content: FF EE DD CC BB AA 99 88 77 66 55 44 33 22 11 00
        int d0 = 0xCCDDEEFF
        int d1 = 0x8899AABB
        int d2 = 0x44556677
        int d3 = 0x00112233

        fh = open_file Test_File {mode} "w"
        assert_result_true()

        write_to_file fh {size} 16 {var_source} d0
        close_file fh

        fh = open_file Test_File {mode} "r"
        assert_result_true()
    return


    :cleanup
        close_file fh
        delete_file {path} Test_File
    return


    // the destination is a raw pointer, so the variable address has to be passed


    function test1(fh: int)
        int buf = 0xcccccccc
        int ptr = get_var_pointer buf

        read_block_from_file fh {size} 0 {address} ptr // tested opcode
        assert_result_true()
        assert_eq(buf, 0xcccccccc) // unchanged
    end


    function test2(fh: int)
        int buf = 0xcccccccc
        int ptr = get_var_pointer buf

        read_block_from_file fh {size} 1 {address} ptr // tested opcode
        assert_result_true()
        assert_eq(buf, 0xccccccFF) // "FF"
    end


    function test3(fh: int)
        int buf = 0xcccccccc
        int ptr = get_var_pointer buf

        read_block_from_file fh {size} 2 {address} ptr // tested opcode
        assert_result_true()
        assert_eq(buf, 0xccccEEFF) // "FF EE"
    end


    function test4(fh: int)
        int buf = 0xcccccccc
        int ptr = get_var_pointer buf

        read_block_from_file fh {size} 3 {address} ptr // tested opcode
        assert_result_true()
        assert_eq(buf, 0xccDDEEFF) // "FF EE DD"
    end


    function test5(fh: int)
        int buf = 0xcccccccc
        int ptr = get_var_pointer buf

        read_block_from_file fh {size} 4 {address} ptr // tested opcode
        assert_result_true()
        assert_eq(buf, 0xCCDDEEFF) // "FF EE DD CC"
    end


    function test6(fh: int)
        int buf  = 0xcccccccc
        int buf2 = 0xcccccccc
        int ptr = get_var_pointer buf

        read_block_from_file fh {size} 6 {address} ptr // tested opcode
        assert_result_true()
        assert_eq(buf, 0xCCDDEEFF)  // "FF EE DD CC"
        assert_eq(buf2, 0xccccAABB) // "AA BB"
    end
end
