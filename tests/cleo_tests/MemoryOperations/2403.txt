{$CLEO .s}
{$INCLUDE_ONCE ../cleo_tester.inc}

script_name "2403" // forget_memory
test("2403 (forget_memory)", tests)
terminate_this_custom_script


function tests
    it("should detach the memory block from the script's allocations", test1)
    return


    function test1
        int addr = allocate_memory {size} 4
        assert_ptr(addr)

        write_memory_with_offset {address} addr {offset} 0 {size} 4 {value} 0x11223344

        forget_memory addr // tested opcode

        // the block is not freed, only detached from CLEO's tracking
        int value = read_memory {address} addr {size} 4 {vp} false
        assert_eq(value, 0x11223344)
    end
end
