{$CLEO .s}
{$INCLUDE_ONCE ../cleo_tester.inc}

script_name "2407" // is_memory_equal
test("2407 (is_memory_equal)", tests)
terminate_this_custom_script


function tests
    it("should compare 0 bytes", test1)
    it("should compare 1 byte", test2)
    it("should compare 2 bytes", test3)
    it("should compare 3 bytes", test4)
    it("should compare 4 bytes", test5)
    it("should compare 8 bytes", test6)
    it("should detect a difference at byte 9", test7)
    it("should detect a difference at byte 16", test8)
    return


    function test1
        int addressA = get_label_pointer @DATA_A
        int addressB = get_label_pointer @DATA_B

        is_memory_equal addressA addressB {size} 0 // tested opcode
        assert_result_true()
    end


    function test2
        int addressA = get_label_pointer @DATA_A
        int addressB = get_label_pointer @DATA_B

        is_memory_equal addressA addressB {size} 1 // tested opcode
        assert_result_true()
    end


    function test3
        int addressA = get_label_pointer @DATA_A
        int addressB = get_label_pointer @DATA_B

        is_memory_equal addressA addressB {size} 2 // tested opcode
        assert_result_true()
    end


    function test4
        int addressA = get_label_pointer @DATA_A
        int addressB = get_label_pointer @DATA_B

        is_memory_equal addressA addressB {size} 3 // tested opcode
        assert_result_true()
    end


    function test5
        int addressA = get_label_pointer @DATA_A
        int addressB = get_label_pointer @DATA_B

        is_memory_equal addressA addressB {size} 4 // tested opcode
        assert_result_true()
    end


    function test6
        int addressA = get_label_pointer @DATA_A
        int addressB = get_label_pointer @DATA_B

        is_memory_equal addressA addressB {size} 8 // tested opcode
        assert_result_true()
    end


    function test7
        int addressA = get_label_pointer @DATA_A
        int addressB = get_label_pointer @DATA_B

        is_memory_equal addressA addressB {size} 9 // tested opcode
        assert_result_false()
    end


    function test8
        int addressA = get_label_pointer @DATA_A
        int addressB = get_label_pointer @DATA_B

        is_memory_equal addressA addressB {size} 16 // tested opcode
        assert_result_false()
    end
end


:DATA_A
hex
    "12345678abcdef" 00
end

:DATA_B
hex
    "1234567812345" 00
end
